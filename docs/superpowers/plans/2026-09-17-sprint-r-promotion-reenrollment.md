# Sprint R — Student Promotion / Re-Enrollment — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development
> (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use
> checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let a `SCHOOL_ADMIN`/`SUPER_ADMIN` move a batch of students from one `AcademicSession` to
the next (promote, retain, transfer out, graduate, or withdraw) without ever mutating an existing
`Enrollment` row in place, and keep a full audit trail of the decision.

**Architecture:** New `backend/src/promotions/` module (service + controller + DTOs), one additive
Prisma migration (`EnrollmentStatus.WITHDRAWN`, new `PromotionDecision` enum, new `StudentPromotion`
model). Every promotion closes the student's current `Enrollment` (sets `status`/`endDate`) and, for
`PROMOTED`/`RETAINED`, creates a brand-new `Enrollment` row for the target session/section — mirrors
the existing `EnrollmentService`/`Enrollment` historical-record pattern from Sprint 6.5 exactly, never
overwrites a prior enrollment. New staff-console `PromotionView.vue` at `/admin/promotions`, plus an
"Academic History" read-only panel on the existing `StudentProfileView.vue`.

**Tech Stack:** NestJS 11 + Prisma 7 (Postgres) on the backend; Vue 3 + TypeScript + Vitest on
staff-console. Follows this codebase's own established `$transaction` + audit-log + tenant-scoping
conventions (see `backend/src/leave/leave.service.ts`, `backend/src/hiring/hiring-applications.service.ts`).

**Spec:** `docs/superpowers/specs/2026-09-17-sprint-r-promotion-reenrollment-design.md`

## Global Constraints

- Every write path is `SCHOOL_ADMIN`/`SUPER_ADMIN` only (`@Roles('SCHOOL_ADMIN', 'SUPER_ADMIN')`,
  matching `LeaveManagementView`'s precedent — `ACCOUNTS` does not get this screen either).
- `execute()` is one `$transaction` per batch call — all-or-nothing, no partial-success reporting
  (no precedent for that shape anywhere in this codebase; validate everything before writing anything,
  matching `LeaveService.approve()`).
- Never call `prisma.enrollment.update()` on `sectionId`/`academicSessionId`/`campusId` of an
  existing row anywhere in this feature — only `status`/`endDate` may be updated on a closed
  enrollment; a new placement is always a new `Enrollment` row.
- `SCHOOL_ADMIN`/`ACCOUNTS`-style tenant scoping: every section referenced (source or target) must
  belong to the acting user's own `schoolId` unless the caller is `SUPER_ADMIN` — mirrors
  `HiringApplicationsService.findMany`'s existing `admin.schoolId` scoping pattern.
- Every state-changing call writes an `AuditLog` row (`action: 'promotion.execute'`), matching every
  other write path in this codebase.

```ts
import { ForbiddenException, Injectable, NotFoundException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import type { RequestUser } from '../common/student-access.service';

export interface PromotionPreviewRow {
  studentId: string;
  name: string;
  grNumber: string;
  currentRollNumber: string | null;
  suggestedDecision: 'PROMOTED';
}

@Injectable()
export class PromotionsService {
  constructor(private readonly prisma: PrismaService) {}

  private async assertSectionInOwnSchool(actingUser: RequestUser, sectionId: string): Promise<void> {
    if (actingUser.role === 'SUPER_ADMIN') {
      return;
    }
    const admin = await this.prisma.user.findUnique({ where: { id: actingUser.id } });
    const section = await this.prisma.section.findUnique({
      where: { id: sectionId },
      select: { class: { select: { campus: { select: { schoolId: true } } } } },
    });
    if (!section) {
      throw new NotFoundException('Section not found');
    }
    if (!admin?.schoolId || section.class.campus.schoolId !== admin.schoolId) {
      throw new ForbiddenException('Cannot access a section outside your own school');
    }
  }

  async preview(actingUser: RequestUser, sourceSectionId: string): Promise<PromotionPreviewRow[]> {
    await this.assertSectionInOwnSchool(actingUser, sourceSectionId);
    const enrollments = await this.prisma.enrollment.findMany({
      where: { sectionId: sourceSectionId, status: 'ACTIVE' },
      include: { student: { select: { name: true, grNumber: true } } },
    });
    return enrollments.map((e) => ({
      studentId: e.studentId,
      name: e.student.name,
      grNumber: e.student.grNumber,
      currentRollNumber: e.rollNumber,
      suggestedDecision: 'PROMOTED' as const,
    }));
  }
}
```

```ts
import { BadRequestException } from '@nestjs/common';
import type { PromotionDecision } from '@prisma/client';

export interface PromotionDecisionInput {
  studentId: string;
  decision: PromotionDecision;
  targetSectionId?: string;
  rollNumber?: string;
  remarks?: string;
}

export interface ExecutePromotionDto {
  sourceAcademicSessionId: string;
  targetAcademicSessionId: string;
  decisions: PromotionDecisionInput[];
}

const CLOSED_STATUS: Record<PromotionDecision, 'COMPLETED' | 'TRANSFERRED' | 'WITHDRAWN'> = {
  PROMOTED: 'COMPLETED',
  RETAINED: 'COMPLETED',
  GRADUATED: 'COMPLETED',
  TRANSFERRED_OUT: 'TRANSFERRED',
  WITHDRAWN: 'WITHDRAWN',
};

const STUDENT_STATUS: Partial<Record<PromotionDecision, 'LEFT' | 'GRADUATED' | 'WITHDRAWN'>> = {
  TRANSFERRED_OUT: 'LEFT',
  GRADUATED: 'GRADUATED',
  WITHDRAWN: 'WITHDRAWN',
};

const CREATES_NEW_ENROLLMENT: PromotionDecision[] = ['PROMOTED', 'RETAINED'];
```

Then add the `execute` method to the `PromotionsService` class:

```ts
  async execute(actingUser: RequestUser, dto: ExecutePromotionDto): Promise<{ processed: number }> {
    for (const item of dto.decisions) {
      if (CREATES_NEW_ENROLLMENT.includes(item.decision) && !item.targetSectionId) {
        throw new BadRequestException(`targetSectionId is required for decision ${item.decision}`);
      }
      if (!CREATES_NEW_ENROLLMENT.includes(item.decision) && item.targetSectionId) {
        throw new BadRequestException(`targetSectionId must not be set for decision ${item.decision}`);
      }
    }

    const targetSession = await this.prisma.academicSession.findUnique({
      where: { id: dto.targetAcademicSessionId },
    });
    if (!targetSession) {
      throw new NotFoundException('Target academic session not found');
    }

    return this.prisma.$transaction(async (tx) => {
      for (const item of dto.decisions) {
        const currentEnrollment = await tx.enrollment.findFirst({
          where: { studentId: item.studentId, status: 'ACTIVE' },
          include: { section: { include: { class: { include: { campus: true } } } } },
        });
        if (!currentEnrollment || currentEnrollment.academicSessionId !== dto.sourceAcademicSessionId) {
          throw new BadRequestException(
            `Student ${item.studentId} has no ACTIVE enrollment in the source academic session`,
          );
        }
        if (actingUser.role !== 'SUPER_ADMIN') {
          const admin = await tx.user.findUnique({ where: { id: actingUser.id } });
          if (!admin?.schoolId || currentEnrollment.section.class.campus.schoolId !== admin.schoolId) {
            throw new ForbiddenException(`Cannot promote student ${item.studentId} outside your own school`);
          }
        }

        await tx.enrollment.update({
          where: { id: currentEnrollment.id },
          data: { status: CLOSED_STATUS[item.decision], endDate: new Date() },
        });

        let newEnrollmentId: string | undefined;
        if (CREATES_NEW_ENROLLMENT.includes(item.decision)) {
          const targetSection = await tx.section.findUnique({
            where: { id: item.targetSectionId },
            include: { class: { include: { campus: true } } },
          });
          if (!targetSection || targetSection.class.academicSessionId !== dto.targetAcademicSessionId) {
            throw new BadRequestException(
              `targetSectionId ${item.targetSectionId} does not belong to the target academic session`,
            );
          }
          if (actingUser.role !== 'SUPER_ADMIN') {
            const admin = await tx.user.findUnique({ where: { id: actingUser.id } });
            if (!admin?.schoolId || targetSection.class.campus.schoolId !== admin.schoolId) {
              throw new ForbiddenException(`Cannot promote student ${item.studentId} outside your own school`);
            }
          }
          const created = await tx.enrollment.create({
            data: {
              studentId: item.studentId,
              campusId: targetSection.class.campusId,
              sectionId: targetSection.id,
              academicSessionId: dto.targetAcademicSessionId,
              startDate: targetSession.startDate,
              status: 'ACTIVE',
              rollNumber: item.rollNumber,
            },
          });
          newEnrollmentId = created.id;
        }

        const studentStatus = STUDENT_STATUS[item.decision];
        if (studentStatus) {
          await tx.student.update({
            where: { id: item.studentId },
            data: { status: studentStatus, leavingDate: new Date(), leavingReason: item.remarks },
          });
        }

        await tx.studentPromotion.create({
          data: {
            studentId: item.studentId,
            fromEnrollmentId: currentEnrollment.id,
            toEnrollmentId: newEnrollmentId,
            decision: item.decision,
            remarks: item.remarks,
            decidedById: actingUser.id,
          },
        });

        await tx.auditLog.create({
          data: {
            userId: actingUser.id,
            action: 'promotion.execute',
            entity: 'Student',
            entityId: item.studentId,
            metadata: JSON.stringify({ decision: item.decision, targetSectionId: item.targetSectionId }),
          },
        });
      }
      return { processed: dto.decisions.length };
    });
  }
```
